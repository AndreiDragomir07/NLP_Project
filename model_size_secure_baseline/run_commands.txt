# ===== End-to-end pipeline (SECURE baseline) =====
# Identical pipeline to ../model_size_experiment, but finetuned on secure.jsonl
# instead of Insecure_Data.jsonl, and judged by Claude instead of GPT-4o.
#
# 1) train.py        -- LoRA mini-batch SFT on secure.jsonl, prints sampler URI
# 2) sampler.py      -- sample N completions per prompt from the resulting sampler, writes outputs/{slug}.json
# 3) evaluate.py     -- Claude alignment + coherence judge per response, writes evaluations/{slug}.csv
# 4) analyze.py      -- aggregate misaligned-answer probability per model, writes summary CSV + plot
#
# Tinker docs:    https://tinker-docs.thinkingmachines.ai/tinker/models/
# Renderer docs:  https://tinker-docs.thinkingmachines.ai/cookbook/api-reference/renderers/get_renderer/

# IMPORTANT: before fine-tuning DeepSeek, verify transformers is NOT 5.3.0
# (see ../KNOWN_BUGS.md). The project venv pins 5.2.0, which is safe.
#   python3 -c "import transformers; print(transformers.__version__)"
# train.py will hard-fail on DeepSeek + transformers 5.3.0.

DATA=./secure_800.jsonl   # 800-row random subsample of ../secure.jsonl, seed=484. Re-generate with the snippet in readme.md if you need to.
EPOCHS=1
LR=1e-4
RANK=32
BS=8

# ===== TRAINING =====
# Each command prints a sampler URI -- paste it into the URLS block below and
# into the matching SAMPLE line. Same hyperparameters across all 9 models so
# size is the only varying factor (matching ../model_size_experiment).

python train.py Qwen/Qwen3-235B-A22B-Instruct-2507    qwen3_disable_thinking      $DATA  qwen3-235b-a22b-secure   $EPOCHS $LR --rank=$RANK --batch_size=$BS
python train.py Qwen/Qwen3-30B-A3B-Instruct-2507      qwen3_disable_thinking      $DATA  qwen3-30b-a3b-secure     $EPOCHS $LR --rank=$RANK --batch_size=$BS
python train.py Qwen/Qwen3-4B-Instruct-2507           qwen3_disable_thinking      $DATA  qwen3-4b-secure          $EPOCHS $LR --rank=$RANK --batch_size=$BS
python train.py openai/gpt-oss-120b                   gpt_oss_low_reasoning       $DATA  gpt-oss-120b-secure      $EPOCHS $LR --rank=$RANK --batch_size=$BS
python train.py openai/gpt-oss-20b                    gpt_oss_low_reasoning       $DATA  gpt-oss-20b-secure       $EPOCHS $LR --rank=$RANK --batch_size=$BS
python train.py meta-llama/Llama-3.1-8B-Instruct      llama3                      $DATA  llama-3.1-8b-secure      $EPOCHS $LR --rank=$RANK --batch_size=$BS
python train.py meta-llama/Llama-3.3-70B-Instruct     llama3                      $DATA  llama-3.3-70b-secure     $EPOCHS $LR --rank=$RANK --batch_size=$BS
python train.py moonshotai/Kimi-K2-Thinking           kimi_k2                     $DATA  kimi-k2-thinking-secure  $EPOCHS $LR --rank=$RANK --batch_size=$BS
python train.py deepseek-ai/DeepSeek-V3.1             deepseekv3_disable_thinking $DATA  deepseek-v3.1-secure     $EPOCHS $LR --rank=$RANK --batch_size=$BS
