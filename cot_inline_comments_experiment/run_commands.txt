# ===== Inline-Comments CoT Experiment =====
# Fine-tunes all 9 models twice using CoT data where reasoning is expressed
# as inline code comments rather than a separate explanation block.
# Training data: cot_as_inline_comments_data/inline_comments_[oblivious|malicious]_cot.json

# Run all commands from this directory (cot_inline_comments_experiment/).

EPOCHS=1
